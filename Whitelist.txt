-- ==================== CONFIGURACIÓN DE SEGURIDAD ====================
local ValidKey = "ERDNONTOP" -- Key escrita en texto
local MainScriptURL = "https://raw.githubusercontent.com/Chrismon014/ERDN/refs/heads/main/PRIVATE" 
local WhitelistURL = "https://raw.githubusercontent.com/Chrismon014/ERDN/refs/heads/main/Key"

local Players = game:GetService("Players")
local LP = Players.LocalPlayer
local PlayerGui = LP:WaitForChild("PlayerGui")

-- ==================== FUNCIÓN DE WHITELIST (CORREGIDA Y BLINDADA) ====================
local function checkWhitelistAndEnforce()
    local success, response = pcall(function()
        return game:HttpGet(WhitelistURL)
    end)

    if not success or not response then
        LP:Kick("\n[ERDN Security]\nError al verificar la Whitelist.\nNo se pudo conectar con el servidor.")
        return false
    end

    local myId = tostring(LP.UserId)
    local myName = LP.Name:lower()
    
    -- Limpieza profunda del texto descargado para evitar errores de saltos de línea
    local isAuthorized = false
    for line in response:gmatch("[^\r\n]+") do
        -- Remueve espacios y caracteres invisibles en cada línea
        local cleanLine = line:gsub("%s+", ""):lower()
        if cleanLine ~= "" then
            if cleanLine == myId or cleanLine == myName then
                isAuthorized = true
                break
            end
        end
    end

    if not isAuthorized then
        -- LO SACA DE ROBLOX SI NO ESTÁ AUTORIZADO
        LP:Kick("\n[ERDN Security Ban]\nNo estás autorizado en la Whitelist.\nAcceso denegado permanentemente.")
        return false
    end

    return true
end

-- Función para cargar el script principal de manera segura
local function loadMainScript()
    pcall(function()
        loadstring(game:HttpGet(MainScriptURL))()
    end)
end

-- ==================== FLUJO PRINCIPAL ====================
if checkWhitelistAndEnforce() then
    
    if PlayerGui:FindFirstChild("ERDNKeySystem") then
        PlayerGui.ERDNKeySystem:Destroy()
    end

    local keyGui = Instance.new("ScreenGui")
    keyGui.Name = "ERDNKeySystem"
    keyGui.ResetOnSpawn = false
    keyGui.IgnoreGuiInset = true
    keyGui.Parent = PlayerGui

    local keyFrame = Instance.new("Frame")
    keyFrame.Size = UDim2.new(0, 300, 0, 180)
    keyFrame.Position = UDim2.new(0.5, -150, 0.5, -90)
    keyFrame.BackgroundColor3 = Color3.fromRGB(10, 10, 15)
    keyFrame.BorderSizePixel = 0
    keyFrame.Parent = keyGui
    Instance.new("UICorner", keyFrame).CornerRadius = UDim.new(0, 12)

    local keyStroke = Instance.new("UIStroke")
    keyStroke.Color = Color3.fromRGB(180, 0, 0)
    keyStroke.Thickness = 2
    keyStroke.Parent = keyFrame

    local keyTitle = Instance.new("TextLabel")
    keyTitle.Size = UDim2.new(1, 0, 0, 40)
    keyTitle.BackgroundTransparency = 1
    keyTitle.Text = "ERDN Key System"
    keyTitle.TextColor3 = Color3.fromRGB(255, 80, 80)
    keyTitle.Font = Enum.Font.GothamBold
    keyTitle.TextSize = 18
    keyTitle.Parent = keyFrame

    local keyInput = Instance.new("TextBox")
    keyInput.Size = UDim2.new(0.85, 0, 0, 40)
    keyInput.Position = UDim2.new(0.075, 0, 0, 60)
    keyInput.BackgroundColor3 = Color3.fromRGB(20, 20, 25)
    keyInput.Text = ""
    keyInput.PlaceholderText = "Enter Key here..."
    keyInput.TextColor3 = Color3.fromRGB(255, 255, 255)
    keyInput.Font = Enum.Font.Gotham
    keyInput.TextSize = 14
    keyInput.Parent = keyFrame
    Instance.new("UICorner", keyInput).CornerRadius = UDim.new(0, 8)

    local verifyBtn = Instance.new("TextButton")
    verifyBtn.Size = UDim2.new(0.85, 0, 0, 40)
    verifyBtn.Position = UDim2.new(0.075, 0, 0, 115)
    verifyBtn.BackgroundColor3 = Color3.fromRGB(180, 30, 30)
    verifyBtn.Text = "Verify Key"
    verifyBtn.TextColor3 = Color3.fromRGB(255, 255, 255)
    verifyBtn.Font = Enum.Font.GothamBold
    verifyBtn.TextSize = 14
    verifyBtn.Parent = keyFrame
    Instance.new("UICorner", verifyBtn).CornerRadius = UDim.new(0, 8)

    verifyBtn.MouseButton1Click:Connect(function()
        if keyInput.Text == ValidKey then
            verifyBtn.Text = "Key Accepted!"
            verifyBtn.BackgroundColor3 = Color3.fromRGB(0, 180, 0)
            task.wait(1)
            keyGui:Destroy()
            
            loadMainScript()
        else
            verifyBtn.Text = "Invalid Key"
            verifyBtn.BackgroundColor3 = Color3.fromRGB(180, 0, 0)
            task.wait(1)
            verifyBtn.Text = "Verify Key"
            verifyBtn.BackgroundColor3 = Color3.fromRGB(180, 30, 30)
        end
    end)
end
